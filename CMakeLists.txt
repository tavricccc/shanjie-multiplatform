function(shanjie_keep_assertions target)
  # CMake's Release flags define NDEBUG; these harnesses use assert() as test assertions.
  target_compile_options(${target} PRIVATE /UNDEBUG)
endfunction()

function(shanjie_harness target source)
  add_executable(${target} ${source})
  target_include_directories(${target} PRIVATE "${SHANJIE_SUPPORT}")
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX)
  target_compile_options(${target} PRIVATE /EHsc /O2 /W4 /utf-8)
  target_link_libraries(${target} PRIVATE ole32 oleaut32 uuid user32 gdi32)
  shanjie_keep_assertions(${target})
endfunction()

function(shanjie_case name case_id)
  add_test(NAME "${name}" COMMAND "${PWSH_EXECUTABLE}" -NoLogo -NoProfile -File "${SHANJIE_CTEST_SCRIPT}" -TestName "${name}" -CaseId "${case_id}" ${ARGN})
  set_tests_properties("${name}" PROPERTIES TIMEOUT 300)
endfunction()

# Two independent runtime slots bound concurrent model loads even if a caller raises --parallel.
set_property(GLOBAL PROPERTY SHANJIE_MODEL_LOCK_INDEX 0)
function(shanjie_assign_model_lock output_variable)
  get_property(index GLOBAL PROPERTY SHANJIE_MODEL_LOCK_INDEX)
  math(EXPR next_index "(${index} + 1) % 2")
  set_property(GLOBAL PROPERTY SHANJIE_MODEL_LOCK_INDEX "${next_index}")
  set(${output_variable} "model-runtime-${index}" PARENT_SCOPE)
endfunction()
